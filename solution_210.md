name: Accessibility Scan

on:
  pull_request:
    branches: [ main, develop ]

    branches: [ main ]

jobs:
  axe:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 18
          cache: 'npm'
      - run: npm ci
      - run: npx playwright install --with-deps
      - name: Typecheck and lint
        run: |
          npm run typecheck
          npm run lint
      - name: Build and start app
        run: |
          npm run build
          npm run start & npx wait-on http://localhost:3000
      - name: Run axe scans
        run: npm run test:e2e -- tests/axe.spec.ts
        env:
          BASE_URL: http://localhost:3000
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: playwright-report
          path: playwright-report/
          retention-days: 30

// tests/axe.spec.ts
import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

const PAGES = ['dashboard', 'send', 'batches', 'payments', 'webhooks'];

test.describe('Axe accessibility scans', () => {
  for (const page of PAGES) {
    test.describe(`Page: ${page}`, () => {
      for (const theme of ['light', 'dark']) {
        test(`should have zero critical/serious violations (${theme})`, async ({ page }) => {
          const baseUrl = process.env.BASE_URL || 'http://localhost:3000';
          await page.goto(`${baseUrl}/${page}`);

          // Set theme via localStorage (adjust key based on your app's implementation)
          await page.evaluate((theme) => {
            localStorage.setItem('theme', theme);
          }, theme);
          await page.reload();

          const axeBuilder = new AxeBuilder({ page });
          const results = await axeBuilder
            .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa'])
            .analyze();

          // Filter violations by impact: only critical and serious are blocking
          const violations = results.violations.filter(
            (v) => v.impact === 'critical' || v.impact === 'serious'
          );

          expect(violations).toEqual([]);
        });
      }
    });
  }
});
