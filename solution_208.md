# Solution for #208: E2E: full batch payment flow (upload → confirm → sign → track)

import { test, expect } from '@playwright/test';

function generateBatchCSV(): string {
  return `address,amount
0x1111111111111111111111111111111111111111,1.0
0x2222222222222222222222222222222222222222,2.5`;
}

test('full batch payment flow (CSV + manual) -> confirm -> sign -> progress -> completion', async ({ page }) => {
  await page.goto('/batches/new');

  const csv = generateBatchCSV();
  await page.setInputFiles('input[type="file"]', {
    name: 'batch.csv',
    mimeType: 'text/csv',
    buffer: Buffer.from(csv),
  });

  await expect(page.locator('[data-testid="batch-item-row"]')).toHaveCount(2);

  await page.click('[data-testid="add-manual-button"]');
  await page.fill('[data-testid="manual-address-input"]', '0x3333333333333333333333333333333333333333');
  await page.fill('[data-testid="manual-amount-input"]', '0.75');
  await page.click('[data-testid="manual-submit-button"]');
  await expect(page.locator('[data-testid="batch-item-row"]')).toHaveCount(3);

  await page.click('[data-testid="validate-button"]');
  await expect(page.locator('[data-testid="validation-badge"]')).toHaveText('Valid');

  await page.click('[data-testid="confirm-button"]');
  await page.click('[data-testid="dialog-confirm-button"]');

  await expect(page.locator('[data-testid="wallet-sign-modal"]')).toBeVisible();
  await page.click('[data-testid="sign-transaction-button"]');

  await expect(page.locator('[data-testid="progress-bar"]')).toHaveAttribute('aria-valuenow', '100');
  await expect(page.locator('[data-testid="batch-status"]')).toHaveText('Completed');
});
